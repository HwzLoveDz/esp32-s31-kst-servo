cmake_minimum_required(VERSION 3.16)
project(kst_fui_preview LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
if(NOT LVGL_SOURCE_DIR)
    message(FATAL_ERROR "Pass -DLVGL_SOURCE_DIR=/absolute/path/to/lvgl-9.5.0")
endif()
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
add_subdirectory("${LVGL_SOURCE_DIR}" "lvgl")
add_executable(kst_fui_preview render.c ../main/fui.c)
target_include_directories(kst_fui_preview PRIVATE ../main)
target_compile_options(kst_fui_preview PRIVATE -Wall -Wextra -Werror)
target_link_libraries(kst_fui_preview PRIVATE lvgl::lvgl m)
